WORD SRAM16CRC(WORD* data)
{
    WORD checksum = 0;
    int rotateValue = 1;
    int rotateCounter = 0;
    int i;
    for(i = 0; i < 0x40; i++)
    {
        checksum = checksum ^ ((data[i] >> rotateValue) | (data[i] << (16 - rotateValue)));
        rotateCounter++;
        if(rotateCounter == 4)
        {
            rotateCounter = 0;
            rotateValue++;
        }
    }
    return checksum;
}
